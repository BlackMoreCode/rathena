//===== rAthena Script =======================================
//= Item Exchange NPC
//= 아이템 교환 NPC
//===== Description: =========================================
//= Accepts predefined items and gives random items from
//= another pool, or allows purchase with Zeny/HPC
//= 미리 정의된 아이템을 받아 다른 풀에서 랜덤 아이템을 주거나
//= Zeny/HPC로 구매 가능
//===== Additional Comments: =================================
//= Based on rAthena scripting conventions
//= rAthena 스크립팅 컨벤션 기준
//============================================================

prontera,160,180,4	script	Item Exchanger	24000,{
	
	// ========================================
	// Main Menu / 메인 메뉴
	// ========================================
	mes "[Item Exchanger]";
	mes "Hello! I can exchange certain items";
	mes "for random rewards from my collection.";
	next;
	
	mes "[Item Exchanger]";
	mes "What would you like to do?";
	next;
	
	switch(select("Exchange items:Buy with Zeny:Buy with HPC:Check exchangeable items:Check reward pool:Cancel")) {
	case 1:
		callsub S_ItemExchange;
		break;
	case 2:
		callsub S_ZenyPurchase;
		break;
	case 3:
		callsub S_HPCPurchase;
		break;
	case 4:
		callsub S_CheckAcceptedItems;
		break;
	case 5:
		callsub S_CheckRewardPool;
		break;
	case 6:
		mes "[Item Exchanger]";
		mes "Come back anytime!";
		close;
	}
	end;

// ========================================
// Sub: Item Exchange / 서브: 아이템 교환
// ========================================
S_ItemExchange:
	mes "[Item Exchanger]";
	mes "I need ^FF00002 items^000000 for one exchange.";
	mes "Choose your exchange method:";
	next;
	
	switch(select("Same item x2:Different items x1 each:Cancel")) {
	case 1:
		callsub S_ExchangeSame;
		break;
	case 2:
		callsub S_ExchangeDifferent;
		break;
	case 3:
		mes "[Item Exchanger]";
		mes "Maybe next time!";
		close;
	}
	end;

// ========================================
// Sub: Exchange Same Item x2 / 서브: 같은 아이템 2개 교환
// ========================================
S_ExchangeSame:
	mes "[Item Exchanger]";
	mes "Checking for items you have at least 2 of...";
	next;
	
	// Build list of available items
	.@count = 0;
	
	for (.@i = 0; .@i < getarraysize(.accepted_items); .@i++) {
		if (countitem(.accepted_items[.@i]) >= 2) {
			.@available_items[.@count] = .accepted_items[.@i];
			.@count++;
		}
	}
	
	if (.@count == 0) {
		mes "[Item Exchanger]";
		mes "You need at least ^FF00002 of the same item^000000.";
		close;
	}
	
	// Pagination settings
	.@items_per_page = 15;
	.@total_pages = (.@count + .@items_per_page - 1) / .@items_per_page;
	.@current_page = 0;
	
	while(1) {
		.@start_idx = .@current_page * .@items_per_page;
		.@end_idx = .@start_idx + .@items_per_page;
		if (.@end_idx > .@count) .@end_idx = .@count;
		
		mes "[Item Exchanger]";
		mes "Which item would you like to exchange?";
		mes "^FF0000(2 of this item will be consumed)^000000";
		if (.@total_pages > 1) {
			mes "^0000FF(Page " + (.@current_page + 1) + "/" + .@total_pages + ")^000000";
		}
		next;
		
		// Build menu with item names only
		.@menu$ = "";
		for (.@i = .@start_idx; .@i < .@end_idx; .@i++) {
			.@menu$ += getitemname(.@available_items[.@i]) + ":";
		}
		
		// Add navigation options
		if (.@total_pages > 1) {
			if (.@current_page > 0) .@menu$ += "Previous Page:";
			if (.@current_page < .@total_pages - 1) .@menu$ += "Next Page:";
		}
		.@menu$ += "Cancel";
		
		.@choice = select(.@menu$) - 1;
		.@page_items = .@end_idx - .@start_idx;
		
		// Handle navigation
		if (.@total_pages > 1) {
			.@nav_start = .@page_items;
			if (.@current_page > 0) {
				if (.@choice == .@nav_start) {
					.@current_page--;
					continue;
				}
				.@nav_start++;
			}
			if (.@current_page < .@total_pages - 1) {
				if (.@choice == .@nav_start) {
					.@current_page++;
					continue;
				}
			}
		}
		
		// Check if cancelled
		if (.@choice >= .@page_items) {
			mes "[Item Exchanger]";
			mes "Maybe next time!";
			close;
		}
		
		.@selected_item = .@available_items[.@start_idx + .@choice];
		break;
	}
	
	// Double-check player still has at least 2 items
	if (countitem(.@selected_item) < 2) {
		mes "[Item Exchanger]";
		mes "You don't have enough of that item anymore.";
		close;
	}
	
	// Show confirmation with owned count
	mes "[Item Exchanger]";
	mes "Selected: ^0000FF" + getitemname(.@selected_item) + "^000000";
	mes "You own: ^00FF00" + countitem(.@selected_item) + "^000000";
	mes " ";
	mes "Exchange ^FF00002^000000 for a random reward?";
	next;
	
	if (select("Yes:No") == 2) {
		mes "[Item Exchanger]";
		mes "Maybe next time!";
		close;
	}
	
	// Remove 2 of the selected item
	delitem .@selected_item, 2;
	
	// Get random item from reward pool
	.@rand = rand(getarraysize(.reward_items));
	getitem .reward_items[.@rand], 1;
	
	mes "[Item Exchanger]";
	mes "Here is your reward:";
	mes "^008800" + getitemname(.reward_items[.@rand]) + "^000000!";
	close;

// ========================================
// Sub: Exchange Different Items / 서브: 다른 아이템 교환
// ========================================
S_ExchangeDifferent:
	mes "[Item Exchanger]";
	mes "Checking your inventory...";
	next;
	
	// Build list of available items
	.@count = 0;
	
	for (.@i = 0; .@i < getarraysize(.accepted_items); .@i++) {
		if (countitem(.accepted_items[.@i]) > 0) {
			.@available_items[.@count] = .accepted_items[.@i];
			.@count++;
		}
	}
	
	if (.@count < 2) {
		mes "[Item Exchanger]";
		mes "You need at least ^FF00002 different items^000000.";
		close;
	}
	
	// Select first item with pagination
	.@items_per_page = 15;
	.@total_pages = (.@count + .@items_per_page - 1) / .@items_per_page;
	.@current_page = 0;
	
	while(1) {
		.@start_idx = .@current_page * .@items_per_page;
		.@end_idx = .@start_idx + .@items_per_page;
		if (.@end_idx > .@count) .@end_idx = .@count;
		
		mes "[Item Exchanger]";
		mes "Select the ^0000FFFIRST item^000000:";
		if (.@total_pages > 1) {
			mes "^0000FF(Page " + (.@current_page + 1) + "/" + .@total_pages + ")^000000";
		}
		next;
		
		.@menu$ = "";
		for (.@i = .@start_idx; .@i < .@end_idx; .@i++) {
			.@menu$ += getitemname(.@available_items[.@i]) + ":";
		}
		
		if (.@total_pages > 1) {
			if (.@current_page > 0) .@menu$ += "Previous Page:";
			if (.@current_page < .@total_pages - 1) .@menu$ += "Next Page:";
		}
		.@menu$ += "Cancel";
		
		.@choice = select(.@menu$) - 1;
		.@page_items = .@end_idx - .@start_idx;
		
		// Handle navigation
		if (.@total_pages > 1) {
			.@nav_start = .@page_items;
			if (.@current_page > 0) {
				if (.@choice == .@nav_start) {
					.@current_page--;
					continue;
				}
				.@nav_start++;
			}
			if (.@current_page < .@total_pages - 1) {
				if (.@choice == .@nav_start) {
					.@current_page++;
					continue;
				}
			}
		}
		
		if (.@choice >= .@page_items) {
			mes "[Item Exchanger]";
			mes "Maybe next time!";
			close;
		}
		
		.@first_item = .@available_items[.@start_idx + .@choice];
		break;
	}
	
	// Build second item list (excluding first item)
	.@count2 = 0;
	
	for (.@i = 0; .@i < .@count; .@i++) {
		if (.@available_items[.@i] == .@first_item) continue;
		.@available_items2[.@count2] = .@available_items[.@i];
		.@count2++;
	}
	
	if (.@count2 == 0) {
		mes "[Item Exchanger]";
		mes "You need a ^FF0000different item^000000.";
		close;
	}
	
	// Select second item with pagination
	.@total_pages2 = (.@count2 + .@items_per_page - 1) / .@items_per_page;
	.@current_page = 0;
	
	while(1) {
		.@start_idx = .@current_page * .@items_per_page;
		.@end_idx = .@start_idx + .@items_per_page;
		if (.@end_idx > .@count2) .@end_idx = .@count2;
		
		mes "[Item Exchanger]";
		mes "First: ^0000FF" + getitemname(.@first_item) + "^000000";
		mes " ";
		mes "Select the ^FF0000SECOND item^000000:";
		if (.@total_pages2 > 1) {
			mes "^0000FF(Page " + (.@current_page + 1) + "/" + .@total_pages2 + ")^000000";
		}
		next;
		
		.@menu$ = "";
		for (.@i = .@start_idx; .@i < .@end_idx; .@i++) {
			.@menu$ += getitemname(.@available_items2[.@i]) + ":";
		}
		
		if (.@total_pages2 > 1) {
			if (.@current_page > 0) .@menu$ += "Previous Page:";
			if (.@current_page < .@total_pages2 - 1) .@menu$ += "Next Page:";
		}
		.@menu$ += "Cancel";
		
		.@choice = select(.@menu$) - 1;
		.@page_items = .@end_idx - .@start_idx;
		
		// Handle navigation
		if (.@total_pages2 > 1) {
			.@nav_start = .@page_items;
			if (.@current_page > 0) {
				if (.@choice == .@nav_start) {
					.@current_page--;
					continue;
				}
				.@nav_start++;
			}
			if (.@current_page < .@total_pages2 - 1) {
				if (.@choice == .@nav_start) {
					.@current_page++;
					continue;
				}
			}
		}
		
		if (.@choice >= .@page_items) {
			mes "[Item Exchanger]";
			mes "Maybe next time!";
			close;
		}
		
		.@second_item = .@available_items2[.@start_idx + .@choice];
		break;
	}
	
	// Double-check player still has both items
	if (countitem(.@first_item) < 1 || countitem(.@second_item) < 1) {
		mes "[Item Exchanger]";
		mes "You don't have the required items anymore.";
		close;
	}
	
	// Confirmation
	mes "[Item Exchanger]";
	mes "You will exchange:";
	mes "^FF0000" + getitemname(.@first_item) + " x1^000000";
	mes "^FF0000" + getitemname(.@second_item) + " x1^000000";
	mes " ";
	mes "For a random reward. Continue?";
	next;
	
	if (select("Yes:No") == 2) {
		mes "[Item Exchanger]";
		mes "Maybe next time!";
		close;
	}
	
	// Remove both items
	delitem .@first_item, 1;
	delitem .@second_item, 1;
	
	// Get random item from reward pool
	.@rand = rand(getarraysize(.reward_items));
	getitem .reward_items[.@rand], 1;
	
	mes "[Item Exchanger]";
	mes "Here is your reward:";
	mes "^008800" + getitemname(.reward_items[.@rand]) + "^000000!";
	close;

// ========================================
// Sub: Zeny Purchase / 서브: Zeny 구매
// ========================================
S_ZenyPurchase:
	mes "[Item Exchanger]";
	mes "I can sell you a random item";
	mes "for ^FF0000" + .zeny_price + " Zeny^000000.";
	next;
	
	mes "[Item Exchanger]";
	mes "Would you like to purchase?";
	next;
	
	if (select("Yes:No") == 2) {
		mes "[Item Exchanger]";
		mes "Maybe next time!";
		close;
	}
	
	if (Zeny < .zeny_price) {
		mes "[Item Exchanger]";
		mes "You don't have enough Zeny.";
		close;
	}
	
	Zeny -= .zeny_price;
	
	.@rand = rand(getarraysize(.reward_items));
	getitem .reward_items[.@rand], 1;
	
	mes "[Item Exchanger]";
	mes "Here is your item:";
	mes "^008800" + getitemname(.reward_items[.@rand]) + "^000000!";
	close;

// ========================================
// Sub: HPC Purchase / 서브: HPC 구매
// ========================================
S_HPCPurchase:
	mes "[Item Exchanger]";
	mes "I can sell you a random item";
	mes "for ^FF0000" + .hpc_price + " HPC^000000.";
	next;
	
	mes "[Item Exchanger]";
	mes "Would you like to purchase?";
	next;
	
	if (select("Yes:No") == 2) {
		mes "[Item Exchanger]";
		mes "Maybe next time!";
		close;
	}
	
	if (countitem(48486) < .hpc_price) {
		mes "[Item Exchanger]";
		mes "You don't have enough HPC.";
		close;
	}
	
	delitem 48486, .hpc_price;
	
	.@rand = rand(getarraysize(.reward_items));
	getitem .reward_items[.@rand], 1;
	
	mes "[Item Exchanger]";
	mes "Here is your item:";
	mes "^008800" + getitemname(.reward_items[.@rand]) + "^000000!";
	close;

// ========================================
// Sub: Check Accepted Items / 서브: 교환 가능 아이템 확인
// ========================================
S_CheckAcceptedItems:
	mes "[Item Exchanger]";
	mes "Here are all the items I can accept for exchange:";
	mes "^0000FF(Total: " + getarraysize(.accepted_items) + " items)^000000";
	next;
	
	.@total = getarraysize(.accepted_items);
	.@page = 0;
	.@items_per_page = 10;
	
	while (.@page * .@items_per_page < .@total) {
		mes "[Item Exchanger]";
		mes "Page " + (.@page + 1) + " / " + ((.@total + .@items_per_page - 1) / .@items_per_page);
		
		.@start = .@page * .@items_per_page;
		.@end = .@start + .@items_per_page;
		if (.@end > .@total) .@end = .@total;
		
		for (.@i = .@start; .@i < .@end; .@i++) {
			.@has_item = countitem(.accepted_items[.@i]);
			if (.@has_item > 0) {
				mes "^00FF00[O]^000000 " + getitemname(.accepted_items[.@i]) + " ^999999(Own: " + .@has_item + ")^000000";
			} else {
				mes "    " + getitemname(.accepted_items[.@i]);
			}
		}
		
		next;
		.@page++;
	}
	
	mes "[Item Exchanger]";
	mes "That's all the items I accept!";
	close;

// ========================================
// Sub: Check Reward Pool / 서브: 보상 풀 확인
// ========================================
S_CheckRewardPool:
	mes "[Item Exchanger]";
	mes "Here are all possible rewards:";
	mes "^0000FF(Total: " + getarraysize(.reward_items) + " items)^000000";
	next;
	
	.@total = getarraysize(.reward_items);
	.@page = 0;
	.@items_per_page = 10;
	
	while (.@page * .@items_per_page < .@total) {
		mes "[Item Exchanger]";
		mes "Page " + (.@page + 1) + " / " + ((.@total + .@items_per_page - 1) / .@items_per_page);
		
		.@start = .@page * .@items_per_page;
		.@end = .@start + .@items_per_page;
		if (.@end > .@total) .@end = .@total;
		
		for (.@i = .@start; .@i < .@end; .@i++) {
			mes "- " + getitemname(.reward_items[.@i]);
		}
		
		next;
		.@page++;
	}
	
	mes "[Item Exchanger]";
	mes "All items have equal chance!";
	close;
